function date(value: unknown): string | null {
  if (value === undefined) return null;
  const text = scalar(value);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(text)) throw new HttpError(400, 'Fecha invÃ¡lida');
  const parsed = new Date(text + 'T00:00:00.000Z');
  if (!Number.isFinite(parsed.getTime()) || parsed.toISOString().slice(0, 10) !== text) throw new HttpError(400, 'Fecha invÃ¡lida');
  return text;
}

    const items = db.all(listSql, [...filters, limit, (page - 1) * limit]);

const LIMITE = 20;

export function ciudadesRouter() {
  const router = Router();
  router.get('/', (req, res) => {
    // Ausente o vacÃ­o = filtro desactivado
    const q = normalizar(scalar(req.query.q) ?? '');
    const country = normalizar(scalar(req.query.country) ?? '');

    const resultado = cities
      .filter(c => {
        const coincideQ =
          !q ||
          [c.ascii, c.country, c.countryCode].some(v => normalizar(v).includes(q));
        const coincidePais =
          !country ||
          [c.country, c.countryCode].some(v => normalizar(v).includes(country));
        return coincideQ && coincidePais; // ambos deben cumplirse
      })
      .slice(0, LIMITE); // primero filtrar, despuÃ©s limitar

    res.json(resultado);
  });
  return router;
}

import { Router } from 'express';
import type { Db } from '../db.js';
import { actor, HttpError, scalar } from '../http.js';

// ---------------------------------------------------------------------------
// REGLAS COMPARTIDAS (una sola definiciÃ³n para listado, contador, lectura,
// ocultamiento y operaciones masivas).
//
// AUDIENCIA: el aviso participa del centro (center=1) y pertenece al usuario si
// es general (broadcast=1), o su rol o su correo figuran como destinatarios.
// EXISTS evita duplicados cuando hay varias coincidencias.
// ParÃ¡metros: [rol, correo]
const AUDIENCIA = `n.center=1 AND (n.broadcast=1 OR EXISTS (
  SELECT 1 FROM destinatarios d WHERE d.notification_id=n.id
  AND ((d.tipo='role' AND d.valor=?) OR (d.tipo='user' AND d.valor=?))))`;

// VISIBLES: audiencia + estado propio del usuario (lecturas filtrada por su
// correo) + no oculta por Ã©l. ParÃ¡metros: [correo, rol, correo]
const VISIBLES = `FROM notificaciones n
  LEFT JOIN lecturas l ON l.notification_id=n.id AND l.user_email=?
  WHERE ${AUDIENCIA} AND l.hidden_at IS NULL`;

type Actor = { role: string; email: string };
const paramsVisibles = (u: Actor) => [u.email, u.role, u.email];

// Una fila en lecturas siempre tiene read_at, asÃ­ que "sin fila" o
// "read_at nulo" significa no leÃ­da.
const NO_LEIDA = 'AND l.read_at IS NULL';

// Garantiza que el aviso existe y pertenece a la audiencia del usuario
// (estÃ© o no oculto por Ã©l). Si no, 404 sin efectos.
function exigirAudiencia(db: Db, u: Actor, id: string) {
  const fila = db.get(
    `SELECT n.id FROM notificaciones n WHERE n.id=? AND ${AUDIENCIA}`,
    [id, u.role, u.email],
  );
  if (!fila) throw new HttpError(404, 'NotificaciÃ³n no encontrada');
}

// Crea la fila propia (leÃ­da ahora) solo si no existÃ­a; nunca pisa fechas.
function asegurarLectura(db: Db, u: Actor, id: string, now: string) {
  db.run(
    'INSERT OR IGNORE INTO lecturas(notification_id,user_email,read_at,hidden_at) VALUES(?,?,?,NULL)',
    [id, u.email, now],
  );
}

// Oculta solo para el usuario, conservando read_at previo y la primera
// fecha de ocultamiento.
function ocultar(db: Db, u: Actor, id: string, now: string) {
  asegurarLectura(db, u, id, now);
  db.run(
    'UPDATE lecturas SET hidden_at=? WHERE notification_id=? AND user_email=? AND hidden_at IS NULL',
    [now, id, u.email],
  );
}

export function notificacionesRouter(db: Db) {
  const router = Router();

  router.get('/', (req, res) => {
    const user = actor(req), value = scalar(req.query.limit, '50');
    if (!/^-?\d+$/.test(value)) throw new HttpError(400, 'LÃ­mite invÃ¡lido');
    const limit = Math.min(100, Math.max(1, Number(value)));

    const items = db.all(
      `SELECT n.id, n.title, n.created_at AS createdAt, l.read_at AS readAt
       ${VISIBLES}
       ORDER BY n.created_at DESC, n.id DESC LIMIT ?`,
      [...paramsVisibles(user), limit],
    );
    // El contador consulta toda la audiencia visible, sin depender del lÃ­mite.
    const conteo = db.get<{ n: number }>(
      `SELECT COUNT(*) AS n ${VISIBLES} ${NO_LEIDA}`,
      paramsVisibles(user),
    );
    res.json({ items, unreadCount: conteo?.n ?? 0 });
  });

  router.patch('/:id/read', (req, res) => {
    const user = actor(req), id = req.params.id, now = new Date().toISOString();
    let readAt = now;
    db.transaction(() => {
      exigirAudiencia(db, user, id);
      asegurarLectura(db, user, id, now); // no sobrescribe ni des-oculta
      const fila = db.get<{ read_at: string }>(
        'SELECT read_at FROM lecturas WHERE notification_id=? AND user_email=?',
        [id, user.email],
      );
      readAt = fila!.read_at; // conserva la primera fecha
    });
    res.json({ id, readAt });
  });

  router.delete('/:id', (req, res) => {
    const user = actor(req), id = req.params.id, now = new Date().toISOString();
    let hiddenAt = now;
    db.transaction(() => {
      exigirAudiencia(db, user, id);
      ocultar(db, user, id, now);
      const fila = db.get<{ hidden_at: string }>(
        'SELECT hidden_at FROM lecturas WHERE notification_id=? AND user_email=?',
        [id, user.email],
      );
      hiddenAt = fila!.hidden_at; // conserva la primera fecha
    });
    res.json({ id, hiddenAt });
  });

  router.post('/read-all', (req, res) => {
    const user = actor(req), now = new Date().toISOString();
    let updated = 0;
    db.transaction(() => {
      const pendientes = db.all<{ id: string }>(
        `SELECT n.id ${VISIBLES} ${NO_LEIDA}`,
        paramsVisibles(user),
      );
      for (const p of pendientes) asegurarLectura(db, user, p.id, now);
      updated = pendientes.length; // solo cambios reales
    });
    res.json({ updated });
  });

  router.post('/clear', (req, res) => {
    const user = actor(req), now = new Date().toISOString();
    let updated = 0;
    db.transaction(() => {
      const visibles = db.all<{ id: string }>(
        `SELECT n.id ${VISIBLES}`,
        paramsVisibles(user),
      );
      for (const v of visibles) ocultar(db, user, v.id, now);
      updated = visibles.length; // solo los que estaban visibles
    });
    res.json({ updated });
  });

  return router;
}

## 1. GET /pagos/historial

Acceso: admin o analista. Un asesor recibe **403**; identidad ausente/desconocida, **401**. La ruta, validaciÃ³n y formato envolvente estÃ¡n implementados; completa las dos consultas SQL.

Filtros opcionales: `fecha_inicio` y `fecha_fin` como fechas de calendario vÃ¡lidas `YYYY-MM-DD`, inclusivas; `cedula_cliente` e `id_credito` como coincidencias exactas, no vacÃ­os. Combinar todos con AND. Rango invertido, fecha imposible, parÃ¡metros repetidos o estructurados: **400**. Las fechas del laboratorio no tienen hora. `page` entero 1..1000000, defecto 1; `limit` entero 1..100, defecto 20. Fuera del rango o no entero: **400**.

Respuesta **200**: `{items,total,page,limit,totalPages}`. Excluir pagos con `deleted_at` no nulo. Orden `fecha_pago DESC,id_pago DESC`. Calcular `total` con todos los filtros antes de paginar. `totalPages = max(1,ceil(total/limit))`. Una pÃ¡gina fuera de rango devuelve items vacÃ­o y conserva total; cero coincidencias devuelve total 0 y totalPages 1.

Cada item contiene exactamente `{id_pago,fecha_pago,id_credito,cedula_cliente,nombre_cliente,estado_credito,metodo_pago,operacion,valor_total}`. Obtener `nombre_cliente` de `clientes` y `estado_credito` de `creditos`; conservar pagos sin esas relaciones, devolviendo los campos relacionados como null. Los IDs de relaciÃ³n de esos pagos tambiÃ©n son null.

`operacion` nula o vacÃ­a se devuelve como `PAGO`. `valor_total` registrado distinto de cero tiene prioridad; si es nulo o cero, sumar `capital`, `interes`, `aval`, `mora`, `cobranza` y `abono_extra`, tratando null como cero.
